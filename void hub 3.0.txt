local function heal_useItem(itemName)
    local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
    if tool and tool.Name == itemName then
        pcall(function()
            tool:Activate()
            task.wait(0.01)
            tool:Deactivate()
        end)
        return true
    end
    local inventory = PlayerGui:FindFirstChild("MainGui")
    if not inventory then return false end
    local rightPanel = inventory:FindFirstChild("RightPanel")
    if not rightPanel then return false end
    local invList = rightPanel:FindFirstChild("Inventory")
    if not invList then return false end
    local list = invList:FindFirstChild("List")
    if not list then return false end
    for _, child in ipairs(list:GetChildren()) do
        if child:IsA("ImageLabel") and child.Name == itemName then
            if packets and packets.UseBagItem and packets.UseBagItem.send then
                packets.UseBagItem.send(child.LayoutOrder)
                return true
            end
        end
    end
    return false
end

-- Цикл Auto Heal
task.spawn(function()
    while true do
        if heal_enabled then
            local currentTime = tick()
            local eatInterval = 1 / heal_cps
            if currentTime - heal_lastEatTime >= eatInterval then
                local humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
                if humanoid then
                    local hpPercent = (humanoid.Health / humanoid.MaxHealth) * 100
                    if hpPercent <= heal_threshold then
                        if heal_useItem(heal_selectedFruit) then
                            heal_lastEatTime = currentTime
                        end
                    end
                end
            end
        end
        task.wait(0.01)
    end
end)

-- Вспомогательные функции Gold Farm
function GoldFarm:getRandomizedSpeed(baseSpeed)
    if tick() - self.LastSpeedChange > self.SpeedChangeInterval then
        if baseSpeed <= 0 then return baseSpeed end
        local minSpeed = baseSpeed - 0.5
        local maxSpeed = baseSpeed
        local chosenSpeed = minSpeed + (math.random() * (maxSpeed - minSpeed))
        self.CurrentSpeedMultiplier = chosenSpeed / baseSpeed
        self.LastSpeedChange = tick()
    end
    return baseSpeed * self.CurrentSpeedMultiplier
end

function GoldFarm:hasLineOfSight(startPos, endPos)
    local direction = (endPos - startPos)
    local distance = direction.Magnitude
    if distance < 1 then return true end
    local rayResult = Workspace:Raycast(startPos + Vector3.new(0, 2, 0), direction, self.RayParams)
    if rayResult then
        local hitDist = (rayResult.Position - startPos).Magnitude
        if hitDist < (distance * 0.9) then return false end
    end
    return true
end

function GoldFarm:wouldTweenHitTerrain(startPos, endPos)
    local sampleCount = 3 
    local direction = (endPos - startPos)
    for i = 0, sampleCount do
        local t = i / sampleCount
        local samplePos = startPos + (direction * t)
        local ray = Workspace:Raycast(samplePos, Vector3.new(0, -5, 0), self.RayParams)
        if not ray then continue end
        if ray and samplePos.Y < ray.Position.Y then return true, samplePos end
    end
    return false
end

function GoldFarm:isPathClearToTarget(startPos, endPos)
    local direction = (endPos - startPos)
    local distance = direction.Magnitude
    if distance < 1 then return true end
    direction = direction.Unit
    local rayOrigin = startPos + Vector3.new(0, 2, 0)
    local rayDirection = (endPos + Vector3.new(0, 2, 0)) - rayOrigin
    local ray = Workspace:Raycast(rayOrigin, rayDirection, self.RayParams)
    if ray then
        local hitDistance = (ray.Position - rayOrigin).Magnitude
        local targetDistance = (endPos - startPos).Magnitude
        if hitDistance < (targetDistance * 0.9) then return false, ray.Position end
    end
    return true
end

function GoldFarm:hasGroundBeneath(position)
    local ray = Workspace:Raycast(position, Vector3.new(0, -100, 0), self.RayParams)
    if ray then
        local hitPart = ray.Instance
        if hitPart and hitPart:IsA("BasePart") then
            local distance = (position - ray.Position).Magnitude
            return true, distance, ray.Position
        end
    end
    return false, 999, nil
end

function GoldFarm:isAboveWater(position)
    if position.Y > 100 then return false end 
    local ray = Workspace:Raycast(position, Vector3.new(0, -200, 0), self.RayParams)
    if ray and ray.Instance and ray.Instance:IsA("Terrain") then
        local region = Region3.new(ray.Position - Vector3.new(4, 4, 4), ray.Position + Vector3.new(4, 4, 4))
        region = region:ExpandToGrid(4)
        local materials, sizes = Workspace.Terrain:ReadVoxels(region, 4)
        local size = materials.Size
        for x = 1, size.X do
            for y = 1, size.Y do
                for z = 1, size.Z do
                    if materials[x][y][z] == Enum.Material.Water then return true end
                end
            end
        end
    end
    return false
end

function GoldFarm:findNearestAccessibleWaypoint(currentPos)
    if #self.RecordedPositions > 0 then
        local closestIndex = self.CurrentWaypointIndex
        local closestDist = math.huge
        local foundAccessible = false
        for idx, pos in ipairs(self.RecordedPositions) do
            if pos and pos.X and pos.Y and pos.Z then
                local waypoint = Vector3.new(pos.X, pos.Y, pos.Z)
                local dist = (waypoint - currentPos).Magnitude
                if dist < closestDist then
                    if dist > 150 then continue end
                    local hasLOS = self:hasLineOfSight(currentPos, waypoint)
                    local wouldHitTerrain = self:wouldTweenHitTerrain(currentPos, waypoint)
                    if hasLOS and not wouldHitTerrain then
                        closestDist = dist
                        closestIndex = idx
                        foundAccessible = true
                    end
                end
            end
        end
        if not foundAccessible then
            closestDist = math.huge
            for idx, pos in ipairs(self.RecordedPositions) do
                if pos and pos.X and pos.Y and pos.Z then
                    local waypoint = Vector3.new(pos.X, pos.Y, pos.Z)
                    local dist = (waypoint - currentPos).Magnitude
                    if dist < closestDist then 
                        closestDist = dist 
                        closestIndex = idx 
                    end
                end
            end
        end
        return closestIndex, closestDist, foundAccessible
    end
    return self.CurrentWaypointIndex, math.huge, false
end

function GoldFarm:findGoldInRadius(position, radius)
    local targets = {}
    local namePatterns = {}
    
    if self.UseVeins then table.insert(namePatterns, "Gold Vein") end
    if self.UseOres then table.insert(namePatterns, "Gold Ore") end
    if self.UseGold then table.insert(namePatterns, "Gold") end
    
    for _, pattern in ipairs(namePatterns) do
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("BasePart") and obj.Name:find(pattern) then
                local distance = (obj.Position - position).Magnitude
                if distance <= radius then
                    local goldAmount = obj:GetAttribute("GoldAmount") or obj:GetAttribute("Amount") or 1
                    if goldAmount >= self.MinGold then
                        table.insert(targets, {
                            Object = obj,
                            Position = obj.Position,
                            Distance = distance,
                            GoldAmount = goldAmount,
                            Name = obj.Name,
                        })
                    end
                end
            end
        end
    end
    
    table.sort(targets, function(a, b) return a.Distance < b.Distance end)
    return targets
end

function GoldFarm:moveToTarget(targetPos)
    local char = LocalPlayer.Character
    if not char then return false end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid then return false end
    
    if self.MovementMode == "Tween" then
        local distance = (targetPos - root.Position).Magnitude
        local speed = self:getRandomizedSpeed(self.WalkSpeed)
        local duration = distance / speed
        
        local tweenInfo = TweenInfo.new(
            duration,
            Enum.EasingStyle.Linear,
            Enum.EasingDirection.Out
        )
        
        local tween = TweenService:Create(root, tweenInfo, {CFrame = CFrame.new(targetPos)})
        tween:Play()
        tween.Completed:Wait()
        return true
    else
        humanoid.AutoRotate = true
        humanoid.WalkSpeed = self:getRandomizedSpeed(self.WalkSpeed)
        
        local startTime = tick()
        local timeout = 30
        
        while tick() - startTime < timeout do
            if not self.Enabled then return false end
            
            local currentPos = root.Position
            local distance = (targetPos - currentPos).Magnitude
            
            if distance < 2 then 
                humanoid.WalkSpeed = 16
                return true 
            end
            
            local direction = (targetPos - currentPos).Unit
            local moveVector = direction * humanoid.WalkSpeed
            
            local pathClear, hitPos = self:isPathClearToTarget(currentPos, targetPos)
            if not pathClear then
                local avoidDirection = (currentPos - hitPos).Unit * Vector3.new(1, 0, 1)
                local avoidPos = currentPos + avoidDirection * 2
                humanoid:MoveTo(avoidPos)
            else
                humanoid:MoveTo(targetPos)
            end
            
            task.wait(0.05)
        end
        
        humanoid.WalkSpeed = 16
        return false
    end
end

function GoldFarm:mineGold(target)
    if not target or not target.Object then return false end
    
    local char = LocalPlayer.Character
    if not char then return false end
    
    local tool = char:FindFirstChildOfClass("Tool")
    if not tool then return false end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return false end
    
    local lookAt = CFrame.new(root.Position, target.Position)
    root.CFrame = lookAt
    
    pcall(function()
        tool:Activate()
        task.wait(0.05)
        tool:Deactivate()
    end)
    
    self.LastNodeTime = tick()
    return true
end

function GoldFarm:updateWaypointVisuals()
    for _, part in pairs(self.WaypointParts) do
        if part and part.Parent then
            part:Destroy()
        end
    end
    self.WaypointParts = {}
    
    for i, waypoint in ipairs(self.Waypoints) do
        local pos = Vector3.new(waypoint.X, waypoint.Y, waypoint.Z)
        local part = self:createWaypointVisual(pos, i)
        table.insert(self.WaypointParts, part)
    end
end

function GoldFarm:addCurrentWaypoint()
    local char = LocalPlayer.Character
    if not char then return false, "Character not found" end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return false, "Root not found" end
    
    local pos = root.Position
    table.insert(self.Waypoints, {
        X = pos.X,
        Y = pos.Y,
        Z = pos.Z
    })
    
    self:updateWaypointVisuals()
    return true, "Waypoint #" .. #self.Waypoints .. " added!"
end

function GoldFarm:clearWaypoints()
    self.Waypoints = {}
    self.WaypointParts = {}
    self.CurrentWaypointIndex = 1
    self:updateWaypointVisuals()
    return true, "All waypoints cleared!"
end

function GoldFarm:removeLastWaypoint()
    if #self.Waypoints > 0 then
        table.remove(self.Waypoints, #self.Waypoints)
        self:updateWaypointVisuals()
        return true, "Last waypoint removed!"
    end
    return false, "No waypoints to remove!"
end

function GoldFarm:loadWaypointsFromString(inputString)
    local success = false
    
    local jsonSuccess, data = pcall(function()
        return HttpService:JSONDecode(inputString)
    end)
    
    if jsonSuccess and type(data) == "table" then
        self:clearWaypoints()
        for i, waypoint in ipairs(data) do
            if waypoint.X and waypoint.Y and waypoint.Z then
                table.insert(self.Waypoints, {
                    X = waypoint.X,
                    Y = waypoint.Y,
                    Z = waypoint.Z
                })
            end
        end
        self:updateWaypointVisuals()
        success = true
    else
        self:clearWaypoints()
        for line in inputString:gmatch("[^\r\n]+") do
            local x, y, z = line:match("([%-%d%.]+)[,%s]+([%-%d%.]+)[,%s]+([%-%d%.]+)")
            if x and y and z then
                table.insert(self.Waypoints, {
                    X = tonumber(x),
                    Y = tonumber(y),
                    Z = tonumber(z)
                })
            end
        end
        if #self.Waypoints > 0 then
            self:updateWaypointVisuals()
            success = true
        end
    end
    
    return success
end

function GoldFarm:copyWaypoints()
    if #self.Waypoints == 0 then return "" end
    
    local data = {}
    for i, w in ipairs(self.Waypoints) do
        table.insert(data, {
            X = w.X,
            Y = w.Y,
            Z = w.Z
        })
    end
    
    local json = HttpService:JSONEncode(data)
    setclipboard(json)
    return json
end

function GoldFarm:moveToNextWaypoint()
    if #self.Waypoints == 0 then return false end
    
    if self.CurrentWaypointIndex > #self.Waypoints then
        self.CurrentWaypointIndex = 1
    end
    
    local waypoint = self.Waypoints[self.CurrentWaypointIndex]
    local targetPos = Vector3.new(waypoint.X, waypoint.Y, waypoint.Z)
    
    local char = LocalPlayer.Character
    if not char then return false end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid then return false end
    
    local currentPos = root.Position
    local distance = (targetPos - currentPos).Magnitude
    
    if distance < 3 then
        self.CurrentWaypointIndex = self.CurrentWaypointIndex + 1
        return true
    end
    
    if self.MovementMode == "Tween" then
        local duration = distance / self.TweenSpeed
        local tween = TweenService:Create(root, TweenInfo.new(duration, Enum.EasingStyle.Linear), {
            CFrame = CFrame.new(targetPos)
        })
        tween:Play()
        tween.Completed:Wait()
    else
        humanoid.AutoRotate = true
        humanoid.WalkSpeed = self.TweenSpeed
        humanoid:MoveTo(targetPos)
        
        local timeout = tick() + 30
        while tick() < timeout do
            if not self.Enabled then break end
            local newDist = (targetPos - root.Position).Magnitude
            if newDist < 3 then break end
            task.wait(0.05)
        end
        humanoid.WalkSpeed = 16
    end
    
    self.CurrentWaypointIndex = self.CurrentWaypointIndex + 1
    return true
end

function GoldFarm:Start()
    if self.Running then return end
    self.Running = true
    self.Enabled = true
    
    task.spawn(function()
        while self.Enabled do
            local char = LocalPlayer.Character
            if not char then 
                task.wait(1)
                continue 
            end
            
            local root = char:FindFirstChild("HumanoidRootPart")
            if not root then 
                task.wait(1)
                continue 
            end
            
            local currentPos = root.Position
            
            if #self.Waypoints > 0 and self.AutoTween then
                if not self:moveToNextWaypoint() then
                    self.CurrentWaypointIndex = 1
                end
                task.wait(0.1)
                continue
            end
            
            local targets = self:findGoldInRadius(currentPos, self.Range)
            
            if #targets > 0 then
                local target = targets[1]
                
                if target.Distance > self.Range then
                    task.wait(0.5)
                    continue
                end
                
                local moved = self:moveToTarget(target.Position)
                
                if moved then
                    self:mineGold(target)
                end
            else
                if #self.RecordedPositions > 0 then
                    local idx, dist = self:findNearestAccessibleWaypoint(currentPos)
                    if dist > 10 then
                        local waypoint = Vector3.new(
                            self.RecordedPositions[idx].X,
                            self.RecordedPositions[idx].Y,
                            self.RecordedPositions[idx].Z
                        )
                        self:moveToTarget(waypoint)
                    end
                else
                    task.wait(1)
                end
            end
            
            task.wait(0.1)
        end
        self.Running = false
    end)
end

function GoldFarm:Stop()
    self.Enabled = false
    self.Running = false
    local char = LocalPlayer.Character
    if char then
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid.WalkSpeed = 16
            humanoid:ChangeState(Enum.HumanoidStateType.Running)
        end
    end
end

local function canExecuteAction()
    local char = LocalPlayer.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end

local function getMyPosition()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    return root and root.Position
end

local function getEquippedToolName()
    local char = LocalPlayer.Character
    local tool = char and char:FindFirstChildOfClass("Tool")
    return tool and tool.Name or ""
end

local function autoDiscoverItemID(toolName)
    return fruitToItemID[toolName] or 0
end

local function sendHitPacket(swingData)
    local p = getgenv().packets or _G.packets
    if p and p.Swing and p.Swing.send then
        p.Swing.send(swingData)
    end
end

local function getWorkspaceResources()
    return Workspace:FindFirstChild("Resources") or Workspace:FindFirstChild("Nodes") or Workspace
end

-- ============================================
-- RESOURCE AURA
-- ============================================
function ResourseAura()
    if not (Settings.Aura and Settings.Aura.Resource) then return end
    if not canExecuteAction() then return end

    local myPos = getMyPosition()
    if not myPos then return end

    local range = Settings.Aura.Range or 20
    local hitLimit = Settings.Aura.ResourceAuraSpeed or 1
    local filterName = Settings.Aura.ResourceFilter or "All"
    local hitsDone = 0

    local toolName = getEquippedToolName()
    local itemID = autoDiscoverItemID(toolName)

    local swingData = {
        toolName = toolName,
        itemID = itemID,
        hitType = "Resource",
        myPos = myPos,
    }

    local resourcesFolder = getWorkspaceResources()
    if not resourcesFolder then return end

    for _, res in ipairs(resourcesFolder:GetChildren()) do
        if hitsDone >= hitLimit then break end

        local resName = res.Name
        local isMatch = (filterName == "All") or (string.lower(resName) == string.lower(filterName))
        if isMatch then
            local part = res:IsA("BasePart") and res or res:FindFirstChildWhichIsA("BasePart", true)
            if part then
                local dist = (part.Position - myPos).Magnitude
                if dist <= range then
                    swingData.targetPart = part
                    swingData.targetPos = part.Position
                    swingData.entityID = res:GetAttribute("EntityID") or res:GetAttribute("ID")

                    sendHitPacket(swingData)
                    hitsDone = hitsDone + 1
                end
            end
        end
    end
end

task.spawn(function()
    while true do
        if Settings.Aura.Resource then
            pcall(ResourseAura)
        end
        task.wait(0.1)
    end
end)

local function farm_getPlantBoxes(range)
    local boxes = {}
    local deployables = Workspace:FindFirstChild("Deployables")
    if not deployables then return boxes end
    
    local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not root then return boxes end
    
    for _, dep in ipairs(deployables:GetChildren()) do
        if dep:IsA("Model") and dep.Name == "Plant Box" then
            local eid = dep:GetAttribute("EntityID")
            local part = dep.PrimaryPart or dep:FindFirstChildWhichIsA("BasePart")
            if eid and part then
                local dist = (part.Position - root.Position).Magnitude
                if dist <= range then
                    table.insert(boxes, { eid = eid, dep = dep, dist = dist })
                end
            end
        end
    end
    return boxes
end

local function farm_getBushes(range, fruit)
    local bushes = {}
    local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not root then return bushes end
    
    for _, model in ipairs(Workspace:GetChildren()) do
        if model:IsA("Model") and model.Name:find(fruit) then
            local part = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
            if part then
                local dist = (part.Position - root.Position).Magnitude
                if dist <= range then
                    local eid = model:GetAttribute("EntityID")
                    if eid then
                        table.insert(bushes, { eid = eid, dist = dist })
                    end
                end
            end
        end
    end
    return bushes
end

local function farm_plantSeed(eid, itemID)
    local p = getgenv().packets or _G.packets
    if p and p.InteractStructure and p.InteractStructure.send then
        p.InteractStructure.send({ entityID = eid, itemID = itemID })
        plantedBoxes[eid] = tick()
    end
end

local function farm_harvestBush(eid)
    local p = getgenv().packets or _G.packets
    if p and p.Pickup and p.Pickup.send then
        p.Pickup.send(eid)
    end
end

-- HITBOX
-- ============================================
local function updateHitboxes()
    for _, v in pairs(Players:GetPlayers()) do
        if v ~= LocalPlayer and v.Character then
            local hrp = v.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                if hitboxEnabled then
                    hrp.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                    
                    for _, child in pairs(hrp:GetChildren()) do
                        if child:IsA("SelectionBox") or child:IsA("BoxHandleAdornment") then
                            child:Destroy()
                        end
                    end
                    
                    local box = Instance.new("SelectionBox")
                    box.Adornee = hrp
                    box.LineThickness = 0.15
                    box.Color3 = Color3.new(1, 0, 0)
                    box.Transparency = 0.5
                    box.Parent = hrp
                else
                    hrp.Size = Vector3.new(2, 2, 1)
                    for _, child in pairs(hrp:GetChildren()) do
                        if child:IsA("SelectionBox") or child:IsA("BoxHandleAdornment") then
                            child:Destroy()
                        end
                    end
                end
            end
        end
    end
end

task.spawn(function()
    while true do
        task.wait(3)
        if hitboxEnabled then
            updateHitboxes()
        end
    end
end)

Players.PlayerAdded:Connect(function(p)

        end
    end)
end)

local function updateSpeed()
    local char = LocalPlayer.Character
    if not char then return end
    
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    
    if SpeedEnabled then
        humanoid.WalkSpeed = TargetWalkSpeed
        humanoid:ChangeState(Enum.HumanoidStateType.Running)
    else
        humanoid.WalkSpeed = DefaultWalkSpeed
    end
end

LocalPlayer.CharacterAdded:Connect(function(char)
    local humanoid = char:WaitForChild("Humanoid")
    task.wait(0.5)
    updateSpeed()
end)

            local humanoid = char:FindFirstChildOfClass("Humanoid")
            if humanoid and humanoid.WalkSpeed ~= TargetWalkSpeed then
                humanoid.WalkSpeed = TargetWalkSpeed
            end
        end
    end
end)

    stroke.Transparency = 0.9
    stroke.Parent = sliderFrame

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.6, 0, 0, 22)
    label.Position = UDim2.new(0, 10, 0, 4)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = COLORS.Text
    label.Font = Enum.Font.GothamMedium
    label.TextSize = 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = sliderFrame
    table.insert(themeElements.Texts, label)

    valueLabel.Size = UDim2.new(0.3, 0, 0, 22)
    valueLabel.Position = UDim2.new(0.7, -10, 0, 4)
    valueLabel.BackgroundTransparency = 1
    valueLabel.Text = tostring(default)
    valueLabel.TextColor3 = COLORS.Accent
    valueLabel.Font = Enum.Font.Gotham
    valueLabel.TextSize = 13
    valueLabel.TextXAlignment = Enum.TextXAlignment.Right
    valueLabel.Parent = sliderFrame
    table.insert(themeElements.Accents, valueLabel)

    local bar = Instance.new("Frame")
    bar.Size = UDim2.new(1, -20, 0, 6)
    bar.Position = UDim2.new(0, 10, 1, -12)
    bar.BackgroundColor3 = Color3.fromRGB(60, 60, 70)
    bar.BorderSizePixel = 0
    bar.Parent = sliderFrame

    local barCorner = Instance.new("UICorner")
    barCorner.CornerRadius = UDim.new(1, 0)
    barCorner.Parent = bar

    local fill = Instance.new("Frame")
    local startPercent = math.clamp((default - min) / (max - min), 0, 1)
    fill.Size = UDim2.new(startPercent, 0, 1, 0)
    fill.BackgroundColor3 = COLORS.Accent
    fill.BorderSizePixel = 0
    fill.Parent = bar
    table.insert(themeElements.Accents, fill)

    local fillCorner = Instance.new("UICorner")
    fillCorner.CornerRadius = UDim.new(1, 0)
    fillCorner.Parent = fill

    local isDragging = false

    local function updateValue(input)
        local barAbsPos = bar.AbsolutePosition.X
        local barAbsSize = bar.AbsoluteSize.X
        local mousePos = input.Position.X
        local percent = math.clamp((mousePos - barAbsPos) / barAbsSize, 0, 1)
        
        local rawVal = min + (max - min) * percent
        local value = math.floor(rawVal + 0.5)

        fill.Size = UDim2.new(percent, 0, 1, 0)
        valueLabel.Text = tostring(value)
        callback(value)
    end

    sliderFrame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isDragging = true
            updateValue(input)
        end
    end)

    sliderFrame.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isDragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateValue(input)
        end
    end)
end

createToggle(miscPage, "Auto Plant", false, function(s)
    farm_autoPlant = s
    if s then
        task.spawn(function()
            while farm_autoPlant do
                local boxes = farm_getPlantBoxes(farm_plantRange)
                local availableBoxes = {}
                
                for _, box in ipairs(boxes) do
                    if not box.dep:FindFirstChild("Seed") then
                        local eid = box.eid
                        if not plantedBoxes[eid] or tick() - plantedBoxes[eid] > 0.5 then
                            table.insert(availableBoxes, box)
                        end
                    end
                end
                
                table.sort(availableBoxes, function(a, b) return a.dist < b.dist end)
                
                for _, box in ipairs(availableBoxes) do
                    if not farm_autoPlant then break end
                    if tick() - lastPlantTime >= farm_plantDelay then
                        farm_plantSeed(box.eid, fruitToItemID[farm_selectedFruit] or 94)
                        lastPlantTime = tick()
                        task.wait(farm_plantDelay)
                    end
                end
                
                for id, time in pairs(plantedBoxes) do
                    if tick() - time > 5 then
                        plantedBoxes[id] = nil
                    end
                end
                
                task.wait(0.05)
            end
        end)
    end
end)

            end
        end)
    end
end)
